#include <string>
#include <vector>
#include <unordered_map>
#include <algorithm>
#include <cmath>

using namespace std;

bool cmp(pair<int, int> p1, pair<int, int> p2)
{
    if(p1.second != p2.second)
        return p1.second > p2.second;
    else
        return p1.first < p2.first;
}

unordered_map<int, int> mDiceNum;

int CheckCase()
{
    if(mDiceNum.size() == 1)
        return 1;
    else if(mDiceNum.size() == 2)
    {
        vector<int> vCount;
        for(auto it = mDiceNum.begin(); it != mDiceNum.end(); ++it)
        {
            vCount.push_back(it->second);
        }
        
        if(vCount[0] == vCount[1])
            return 3;
        else
            return 2;
    }
    else if(mDiceNum.size() == 3)
        return 4;
    
    return 5;
}

int solution(int a, int b, int c, int d) {

    int nScore = 0;
    int nCase;
    int p, q, r;
    vector<pair<int, int>> vCount;
    mDiceNum[a]++;
    mDiceNum[b]++;
    mDiceNum[c]++;
    mDiceNum[d]++;
    
   
    for(auto it = mDiceNum.begin(); it != mDiceNum.end(); ++it)
    {
        vCount.push_back({it->first, it->second});
    }
    sort(vCount.begin(), vCount.end(), cmp);
    
    nCase = CheckCase();

    switch(nCase)
    {
        case 1:
            nScore = 1111 * a;
            break;
        case 2:

            p = vCount[0].first;
            q = vCount[1].first;
            
            nScore = (10 * p + q) * (10 * p + q);
            break;
        case 3:
            p = vCount[0].first;
            q = vCount[1].first;
            
            nScore = (p + q) * abs(p - q);
            break;
        case 4:
            q = vCount[1].first;
            r = vCount[2].first;
            
            nScore = q * r; 
            break;
        case 5:
            nScore = vCount[0].first;
            break;
    }
    
    return nScore;
    
}
