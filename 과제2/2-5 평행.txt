#include <string>
#include <vector>

using namespace std;

vector<vector<int>> vDots;

bool Cmp(int i, int j)
{
    float nSlope1, nSlope2;
    vector<vector<int>> vRemain;
    
    nSlope1 = (float)(vDots[i][1] - vDots[j][1]) / (float)(vDots[i][0] - vDots[j][0]);
    
    for(int k = 0; k < vDots.size(); ++k)
    {
        if(k != i && k != j)
            vRemain.push_back(vDots[k]);
    }
    
    nSlope2 = (float)(vRemain[0][1] - vRemain[1][1]) / (float)(vRemain[0][0] - vRemain[1][0]);
    
    if(nSlope1 == nSlope2)
        return true;
    else
        return false;
}

int solution(vector<vector<int>> dots) {
    
    vDots = dots;
    
    for(int i = 0; i < dots.size(); ++i)
    {
        for(int j = i + 1; j < dots.size(); ++j)
        {
            if(Cmp(i, j))
                return 1;
        }
    }
    
    return 0;
}
